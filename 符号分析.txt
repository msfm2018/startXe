
#include <windows.h>
#include <iostream>
#include <string>
#include <DbgHelp.h>

#pragma comment(lib, "dbghelp.lib")

typedef int (*AbcFunc)();

// 回调：用于枚举符号时查找 abc 函数
BOOL CALLBACK EnumSymbolsCallback(PSYMBOL_INFO pSymInfo, ULONG SymbolSize, PVOID UserContext) {
    const char* targetName = static_cast<const char*>(UserContext);

    if (strcmp(pSymInfo->Name, targetName) == 0) {
        std::wcout << L"[枚举找到符号] '" << targetName << L"' 地址: 0x" << std::hex << pSymInfo->Address << std::endl;

        // 可在此中止枚举（找到了）
        AbcFunc func = reinterpret_cast<AbcFunc>(pSymInfo->Address);
        int result = func();
        std::wcout << L"abc() 返回: " << std::dec << result << std::endl;

        return FALSE; // 停止枚举
    }

    return TRUE; // 继续
}

int main() {
    std::wcout.imbue(std::locale(""));

    // 初始化符号处理器
    if (!SymInitialize(GetCurrentProcess(), NULL, TRUE)) {
        std::wcerr << L"[错误] SymInitialize 失败：" << GetLastError() << std::endl;
        return 1;
    }

    // 加载 DLL（确保路径正确）
    const wchar_t* dllPath = L"C:\\aa\\Project2.dll";
    HMODULE hMod = LoadLibraryW(dllPath);
    if (!hMod) {
        std::wcerr << L"[错误] 无法加载 DLL：" << GetLastError() << std::endl;
        SymCleanup(GetCurrentProcess());
        return 1;
    }

    // 加载符号信息
    DWORD64 baseAddress = SymLoadModule64(
        GetCurrentProcess(),
        NULL,
        "C:\\a\\Project2.dll",  // ANSI 路径
        NULL,
        (DWORD64)hMod,
        0
    );

    if (baseAddress == 0) {
        std::wcerr << L"[错误] SymLoadModule64 失败：" << GetLastError()
            << L"（确保 PDB 文件匹配）" << std::endl;
        FreeLibrary(hMod);
        SymCleanup(GetCurrentProcess());
        return 1;
    }

    std::wcout << L"[信息] 成功加载符号，模块基址: 0x" << std::hex << baseAddress << std::endl;

    // 使用 SymFromName 直接查找符号（假设 C 函数名为 abc）
    const char* symbolName = "abc"; // 如果是 C++ 函数，用混淆名，如 "?abc@@YAHXZ"

    SYMBOL_INFO_PACKAGE sip = {};
    sip.si.SizeOfStruct = sizeof(SYMBOL_INFO);
    sip.si.MaxNameLen = MAX_SYM_NAME;

    if (SymFromName(GetCurrentProcess(), symbolName, &sip.si)) {
        std::wcout << L"[找到符号] '" << symbolName << L"' 地址: 0x"
            << std::hex << sip.si.Address << std::endl;

        AbcFunc abcFunc = reinterpret_cast<AbcFunc>(sip.si.Address);
        if (abcFunc) {
            int ret = abcFunc();
            std::wcout << L"abc() 返回值: " << std::dec << ret << std::endl;
        }
    }
    else {
        std::wcerr << L"[警告] SymFromName 未找到符号 '" << symbolName
            << L"'，错误：" << GetLastError() << std::endl;
        std::wcerr << L"提示：可能是 C++ 函数，尝试使用混淆名或枚举所有符号。" << std::endl;
    }

    // 枚举所有符号作为备选方案
    std::wcout << L"\n[枚举符号] 开始查找 abc..." << std::endl;
    if (!SymEnumSymbols(GetCurrentProcess(), baseAddress, NULL, EnumSymbolsCallback, (PVOID)symbolName)) {
        std::wcerr << L"[错误] SymEnumSymbols 失败：" << GetLastError() << std::endl;
    }

    // 清理资源
    SymUnloadModule64(GetCurrentProcess(), baseAddress);
    FreeLibrary(hMod);
    SymCleanup(GetCurrentProcess());

    std::wcout << L"\n程序结束。" << std::endl;
    return 0;
}
